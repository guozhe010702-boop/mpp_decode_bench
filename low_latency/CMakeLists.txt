cmake_minimum_required(VERSION 3.16)
project(imx415_low_latency LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SDK_PREFIX /userdata/av_dev/ffmpeg_pipeline/runtime CACHE PATH "Private board SDK")
find_package(PkgConfig REQUIRED)
pkg_check_modules(DRM REQUIRED libdrm)
find_library(MPP rockchip_mpp PATHS ${SDK_PREFIX}/lib NO_DEFAULT_PATH REQUIRED)
add_executable(imx415_low_latency main.cpp)
target_include_directories(imx415_low_latency PRIVATE ${SDK_PREFIX}/include/rockchip ${DRM_INCLUDE_DIRS})
target_link_libraries(imx415_low_latency PRIVATE ${MPP} ${DRM_LIBRARIES} pthread)
target_compile_options(imx415_low_latency PRIVATE -Wall -Wextra -Wpedantic -O2)
set_target_properties(imx415_low_latency PROPERTIES BUILD_RPATH "${SDK_PREFIX}/lib")
